Var_Func(src, length) =>
    valpha = 2 / (length + 1)
    vud1 = src > src[1] ? src - src[1] : 0
    vdd1 = src < src[1] ? src[1] - src : 0
    vUD = math.sum(vud1, 9)
    vDD = math.sum(vdd1, 9)
    vCMO = nz((vUD - vDD) / (vUD + vDD))
    VAR = 0.0
    VAR := nz(valpha * math.abs(vCMO) * src) + (1 - valpha * math.abs(vCMO)) * nz(VAR[1])
    VAR

Wwma_Func(src, length) =>
    wwalpha = 1 / length
    WWMA = 0.0
    WWMA := wwalpha * src + (1 - wwalpha) * nz(WWMA[1])
    WWMA

Zlema_Func(src, length) =>
    zxLag = length / 2 == math.round(length / 2) ? length / 2 : (length - 1) / 2
    zxEMAData = src + src - src[zxLag]
    ZLEMA = ta.ema(zxEMAData, length)
    ZLEMA

Tsf_Func(src, length) =>
    lrc = ta.linreg(src, length, 0)
    lrc1 = ta.linreg(src, length, 1)
    lrs = lrc - lrc1
    TSF = ta.linreg(src, length, 0) + lrs
    TSF

// AVANT: Short seulement si corrélation <= -threshold (très restrictif, rare)
// APRÈS: Short autorisé si corrélation est significative (positive OU négative)
//        car une forte corrélation (même positive) signifie que l'actif suit le marché
//        et donc les signaux short du système sont valides
// LOGIQUE:
//   - Corrélation >= +threshold : Actif suit le marché → Long ET Short OK
//   - Corrélation <= -threshold : Actif inverse du marché → Long ET Short OK (inversés)
//   - Corrélation entre -threshold et +threshold : Actif décorrélé → Prudence
getCorrelation(ref_data) =>
    if na(ref_data)
        [true, true]  // Filter disabled = authorize all trades
    else
        // Main instrument returns
        returns_main = (close - close[1]) / close[1]
        
        // Reference asset returns (using pre-fetched data)
        // V68.1: Protection division par zéro
        returns_ref = ref_data[1] != 0 ? (ref_data - ref_data[1]) / ref_data[1] : 0
        
        // Calculate correlation
        mean_main = ta.sma(returns_main, corr_length)
        mean_ref = ta.sma(returns_ref, corr_length)
        
        numerator = ta.sma((returns_main - mean_main) * (returns_ref - mean_ref), corr_length)
        denominator = ta.stdev(returns_main, corr_length) * ta.stdev(returns_ref, corr_length)
        
        correlation = denominator != 0 ? numerator / denominator : 0
        
        // V68.7.1 FIX: Nouvelle logique basée sur la FORCE de corrélation
        // Si la corrélation est forte (positive OU négative), l'actif est prévisible
        // → On autorise les trades dans les deux directions
        // Si la corrélation est faible (proche de 0), l'actif est imprévisible
        // → On bloque les trades
        abs_correlation = math.abs(correlation)
        is_correlated = abs_correlation >= corr_threshold
        
        // Long: Autorisé si actif est corrélé au marché (positivement ou négativement)
        // Short: Autorisé si actif est corrélé au marché (positivement ou négativement)
        [is_correlated, is_correlated]
